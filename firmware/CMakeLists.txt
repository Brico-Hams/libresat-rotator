cmake_minimum_required(VERSION 3.13)
# LibreSat Rotator firmware (Raspberry Pi Pico / Pico W / Pico 2)
#   export PICO_SDK_PATH=/path/to/pico-sdk   (2.x)
#   cmake -B build -DPICO_BOARD=pico && cmake --build build -j                     -> build/lsr1.uf2
#   cmake -B build-lsr2 -DLSR_MODEL=2 && cmake --build build-lsr2 -j              -> build-lsr2/lsr2.uf2
# LSR_MODEL=1|2 (default 1); LSR_DRIVER=TMC2209_UART|EXT_STEPDIR (default: EXT_STEPDIR for model 2)
if(NOT DEFINED LSR_MODEL)
  set(LSR_MODEL 1)
endif()
include(pico_sdk_import.cmake)
project(lsr${LSR_MODEL} C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()

set(TGT lsr${LSR_MODEL})
add_executable(${TGT} src/main.cpp src/core.cpp src/hw.cpp)
pico_generate_pio_header(${TGT} ${CMAKE_CURRENT_LIST_DIR}/src/tmc_uart.pio)
target_compile_definitions(${TGT} PRIVATE LSR_MODEL=${LSR_MODEL})
if(DEFINED LSR_DRIVER)
  target_compile_definitions(${TGT} PRIVATE LSR_DRIVER=${LSR_DRIVER})
endif()
target_compile_options(${TGT} PRIVATE -Wall -Wno-unused-function -Wno-misleading-indentation)
target_link_libraries(${TGT} pico_stdlib hardware_pio hardware_i2c hardware_uart hardware_adc
                      hardware_flash hardware_sync hardware_watchdog)
pico_enable_stdio_usb(${TGT} 1)
pico_enable_stdio_uart(${TGT} 0)
pico_set_program_name(${TGT} "LibreSat Rotator LSR-${LSR_MODEL}")
pico_set_program_version(${TGT} "1.0.0")
pico_add_extra_outputs(${TGT})
